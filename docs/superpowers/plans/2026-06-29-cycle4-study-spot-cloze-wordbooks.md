  /// Great-circle distance in meters between two lat/lng points (haversine).
  static double distanceMeters(
      double lat1, double lng1, double lat2, double lng2) {
    const earthRadius = 6371000.0;
    final dLat = _radians(lat2 - lat1);
    final dLng = _radians(lng2 - lng1);
    final a = sin(dLat / 2) * sin(dLat / 2) +
        cos(_radians(lat1)) *
            cos(_radians(lat2)) *
            sin(dLng / 2) *
            sin(dLng / 2);
    return earthRadius * 2 * atan2(sqrt(a), sqrt(1 - a));
  }

  static double _radians(double deg) => deg * pi / 180;

  /// Index of the nearest spot within [withinMeters] meters of
  /// ([latitude], [longitude]), or null if none qualify.
  static int? nearestWithin(
    List<({double latitude, double longitude})> spots,
    double latitude,
    double longitude, {
    double withinMeters = 100,
  }) {
    int? best;
    var bestDist = withinMeters;
    for (var i = 0; i < spots.length; i++) {
      final d = distanceMeters(
          latitude, longitude, spots[i].latitude, spots[i].longitude);
      if (d <= bestDist) {
        bestDist = d;
        best = i;
      }
    }
    return best;
  }
```

  /// Nearby upsert: if an existing spot is within 100 m, increment its
  /// visit_count and bump created_at; otherwise insert a new spot
  /// (visit_count = 1). Returns the affected spot.
  Future<Map<String, dynamic>> recordStudyVisit(
    String userId, {
    required String name,
    required double latitude,
    required double longitude,
  }) async {
    final existing = await connection.execute(
      Sql.named(
        'SELECT id, latitude, longitude FROM study_spots WHERE user_id=@userId',
      ),
      parameters: {'userId': userId},
    );
    final spots = existing
        .map((row) => (
              id: row[0].toString(),
              latitude: (row[1] as num).toDouble(),
              longitude: (row[2] as num).toDouble(),
            ))
        .toList();
    final idx = StudySpotPolicy.nearestWithin(
      spots.map((s) => (latitude: s.latitude, longitude: s.longitude)).toList(),
      latitude,
      longitude,
    );
    final returning =
        'RETURNING id, name, latitude, longitude, study_minutes, visit_count, created_at';
    final result = idx != null
        ? await connection.execute(
            Sql.named(
              'UPDATE study_spots SET visit_count = visit_count + 1, created_at = NOW() '
              'WHERE id=@id::uuid AND user_id=@userId $returning',
            ),
            parameters: {'id': spots[idx].id, 'userId': userId},
          )
        : await connection.execute(
            Sql.named(
              'INSERT INTO study_spots (user_id, name, latitude, longitude, visit_count) '
              'VALUES (@userId, @name, @lat, @lng, 1) $returning',
            ),
            parameters: {
              'userId': userId,
              'name': name,
              'lat': latitude,
              'lng': longitude,
            },
          );
    final row = result.first;
    return {
      'id': row[0].toString(),
      'name': row[1],
      'latitude': row[2],
      'longitude': row[3],
      'studyMinutes': row[4],
      'visitCount': row[5],
      'createdAt': row[6].toString(),
    };
  }

import 'package:geolocator/geolocator.dart';

/// Returns the device's current position, or null when permission is denied
/// or location is otherwise unavailable. Never throws — callers treat null as
/// "skip silently".
Future<LatLng?> currentLatLng() async {
  try {
    var perm = await Geolocator.checkPermission();
    if (perm == LocationPermission.denied) {
      perm = await Geolocator.requestPermission();
    }
    if (perm == LocationPermission.denied ||
        perm == LocationPermission.deniedForever) {
      return null;
    }
    final pos = await Geolocator.getCurrentPosition();
    return LatLng(pos.latitude, pos.longitude);
  } catch (_) {
    return null;
  }
}
```

  @override
  Future<StudySpot> recordVisit({
    required double latitude,
    required double longitude,
    required String name,
  }) async {
    final prefs = await SharedPreferences.getInstance();
    final spots = await _read(prefs);
    final distance = const Distance();
    final here = LatLng(latitude, longitude);
    var nearestIndex = -1;
    var nearestMeters = 100.0; // ≤100m aggregates
    for (var i = 0; i < spots.length; i++) {
      final d = distance.as(
          LengthUnit.Meter, here, LatLng(spots[i].latitude, spots[i].longitude));
      if (d <= nearestMeters) {
        nearestMeters = d;
        nearestIndex = i;
      }
    }
    final StudySpot saved;
    if (nearestIndex >= 0) {
      final old = spots[nearestIndex];
      saved = StudySpot(
        id: old.id,
        name: old.name,
        latitude: old.latitude,
        longitude: old.longitude,
        studyMinutes: old.studyMinutes,
        visitCount: old.visitCount + 1,
        createdAt: DateTime.now().toUtc(),
      );
      spots[nearestIndex] = saved;
    } else {
      saved = StudySpot(
        id: 'spot-${DateTime.now().microsecondsSinceEpoch}',
        name: name,
        latitude: latitude,
        longitude: longitude,
        studyMinutes: 0,
        visitCount: 1,
        createdAt: DateTime.now().toUtc(),
      );
      spots.add(saved);
    }
    await _write(prefs, spots);
    return saved;
  }
```

  /// Records a study visit (nearby upsert via the repository). When [latitude]/
  /// [longitude] are omitted, the current device location is used. Any
  /// failure — denied permission, timeout, network — is swallowed so it never
  /// blocks lesson/training completion.
  Future<void> recordStudyVisit({
    required String name,
    double? latitude,
    double? longitude,
  }) async {
    try {
      double lat;
      double lng;
      if (latitude != null && longitude != null) {
        lat = latitude;
        lng = longitude;
      } else {

#!/usr/bin/env python3
"""Build the full server HSK wordbook file from the client asset books.

Reads linguaquest/lib/assets/data/wordbook_hsk{1..6}.json (each a single book
object) and writes server/data/wordbooks_p0.json as a 6-element array,
replacing the stub hsk2-6 with the full client lists. ids are unchanged.

Run from the repo root (ASM2/):
    python3 tools/sync_p0_from_assets.py
"""
import json
import os

REPO = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
ASSETS = os.path.join(REPO, "linguaquest", "lib", "assets", "data")
OUT = os.path.join(REPO, "server", "data", "wordbooks_p0.json")


def main():
    books = []
    for level in range(1, 7):
        path = os.path.join(ASSETS, f"wordbook_hsk{level}.json")
        with open(path, encoding="utf-8") as f:
            book = json.load(f)
        assert book["id"] == f"hsk{level}", book["id"]
        assert len(book["words"]) == len(set(w["id"] for w in book["words"])), (
            f"dup ids in hsk{level}"
        )
        books.append(book)
    with open(OUT, "w", encoding="utf-8") as f:
        json.dump(books, f, ensure_ascii=False, indent=2)
    print(f"wrote {OUT}: " + ", ".join(
        f"{b['id']}={len(b['words'])}" for b in books))


if __name__ == "__main__":
    main()
```

def slug(word, i):
    ascii_word = "".join(c for c in word if c.isascii() and (c.isalnum()))
    return f"{i:03d}_{ascii_word.lower()[:8] or 'w'}"


def youdao(word, t, le):
    q = urllib.parse.quote(word, safe="")
    return f"https://dict.youdao.com/dictvoice?audio={q}&type={t}&le={le}"


def make_word(lang, word, gloss, i):
    le = AUDIO_LE.get(lang)
    return {
        "id": f"{lang}_{slug(word, i)}",
        "word": word,
        "phoneticUK": "",
        "phoneticUS": "",
        "audioUKUrl": youdao(word, 1, le) if le else "",
        "audioUSUrl": youdao(word, 2, le) if le else "",
        "partOfSpeech": "",
        "definitions": [gloss],
        "englishDefinitions": [gloss],
        "malayDefinitions": [],
        "exampleSentences": [],
        "collocations": [],
        "synonyms": [],
        "antonyms": [],
        "derivatives": [],
        "rootAffix": "",
        "examTags": [],
        "difficulty": "Easy",
        "imageUrl": "",
    }


def make_book(lang, meta):
    words = [make_word(lang, w, g, i) for i, (w, g) in enumerate(meta["words"])]
    ids = [w["id"] for w in words]
    assert len(ids) == len(set(ids)), f"dup ids in {lang}"
    title = meta["title"]
    return {
        "id": f"{lang}_basic",
        "title": title,
        "titleEn": title,
        "titleMs": meta["titleMs"],
        "category": "basic",
        "examType": "Basic",
        "coverUrl": "",
        "language": lang,
        "totalWords": len(words),
        "includedWords": len(words),
        "description": f"Everyday {title} vocabulary (test set).",
        "wordIds": ids,
        "createdAt": CREATED_AT,
        "words": words,
    }


def main():
    books = [make_book(lang, meta) for lang, meta in BOOKS.items()]
    with open(SERVER_OUT, "w", encoding="utf-8") as f:
        json.dump(books, f, ensure_ascii=False, indent=2)
    print(f"wrote {SERVER_OUT}: {len(books)} books")
    for book in books:
        path = os.path.join(ASSETS, f"wordbook_{book['id']}.json")
        with open(path, "w", encoding="utf-8") as f:
            json.dump(book, f, ensure_ascii=False, indent=2)
        print(f"wrote {path}: {len(book['words'])} words")


if __name__ == "__main__":
    main()
