# HSK Target-Language Flip Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Flip the app from *Chinese-speakers-learn-English* to *English-speakers-learn-Chinese (HSK)*: lock the UI to English, add a target-language selector (Chinese active; Spanish/French/Japanese coming-soon), replace the English wordbooks with an HSK band-1 seed, and flip quiz prompts/answers to the new direction.

**Architecture:** Reuse the existing rich `Word` model (one new optional field `hskLevel`); populate `definitions`/`englishDefinitions` with English meanings and `phoneticUK/US` with pinyin so the existing `definitionsFor('en')` and quiz machinery work unchanged. Seed data is generated by a committed one-off Dart converter from `drkameleon/complete-hsk-vocabulary`. UI locale is hard-forced to English; the old language picker becomes a target-language picker reusing the existing coming-soon pattern.

**Tech Stack:** Flutter, Riverpod, Dart. Spec: `docs/superpowers/specs/2026-06-18-hsk-target-language-flip-design.md`.

**All commands run from `linguaquest/` unless noted.**

```dart
/// Normalizes pinyin for lenient answer comparison: strips tone diacritics,
/// lowercases, maps ü/v → u, and removes spaces, apostrophes and hyphens.
///
/// Users typing pinyin on a standard keyboard can't produce tone marks or the
/// `ü` glyph, so spelling answers are compared on this normalized form.
String normalizePinyin(String input) {
  final buffer = StringBuffer();
  for (final rune in input.toLowerCase().runes) {
    final char = String.fromCharCode(rune);
    final mapped = _toneless[char];
    if (mapped != null) {
      buffer.write(mapped);
      continue;
    }
    if (char == 'v' || char == 'ü') {
      buffer.write('u');
      continue;
    }
    if (RegExp(r'[a-z]').hasMatch(char)) {
      buffer.write(char);
    }
    // Everything else (spaces, apostrophes, hyphens, digits, punctuation) is
    // dropped.
  }
  return buffer.toString();
}

const Map<String, String> _toneless = {
  'ā': 'a', 'á': 'a', 'ǎ': 'a', 'à': 'a',
  'ē': 'e', 'é': 'e', 'ě': 'e', 'è': 'e',
  'ī': 'i', 'í': 'i', 'ǐ': 'i', 'ì': 'i',
  'ō': 'o', 'ó': 'o', 'ǒ': 'o', 'ò': 'o',
  'ū': 'u', 'ú': 'u', 'ǔ': 'u', 'ù': 'u',
  'ǖ': 'u', 'ǘ': 'u', 'ǚ': 'u', 'ǜ': 'u', 'ü': 'u',
  'ń': 'n', 'ň': 'n', 'ǹ': 'n', 'ḿ': 'm',
};
```

Create `tools/hsk_seed/convert_hsk.dart`:

```dart
// One-off seed generator. NOT part of the app build.
//
// Usage:
//   curl -fsSL \
//     https://raw.githubusercontent.com/drkameleon/complete-hsk-vocabulary/main/wordlists/exclusive/new/1.json \
//     -o /tmp/hsk_new_1.json
//   dart run tools/hsk_seed/convert_hsk.dart /tmp/hsk_new_1.json \
//     linguaquest/lib/assets/data/wordbook_hsk1.json
//
// Source entry shape (drkameleon/complete-hsk-vocabulary):
//   { "simplified": "爱", "pos": ["v"], "frequency": 130,
//     "forms": [ { "traditional": "愛",
//                  "transcriptions": { "pinyin": "ài", ... },
//                  "meanings": ["to love", ...],
//                  "classifiers": [] } ] }
import 'dart:convert';
import 'dart:io';

const _hskLevel = 1;
// Fixed timestamp keeps the generated asset deterministic across runs.
const _createdAt = '2026-06-19T00:00:00.000Z';

void main(List<String> args) {
  if (args.length != 2) {
    stderr.writeln('Usage: dart run tools/hsk_seed/convert_hsk.dart <input.json> <output.json>');
    exit(64);
  }
  final raw = jsonDecode(File(args[0]).readAsStringSync());
  if (raw is! List) {
    stderr.writeln('Expected a JSON array at top level.');
    exit(65);
  }

  final words = <Map<String, dynamic>>[];
  final usedIds = <String>{};
  for (var i = 0; i < raw.length; i++) {
    final entry = Map<String, dynamic>.from(raw[i] as Map);
    final simplified = (entry['simplified'] ?? '').toString();
    if (simplified.isEmpty) continue;
    final forms = (entry['forms'] as List?) ?? const [];
    if (forms.isEmpty) continue;
    final form = Map<String, dynamic>.from(forms.first as Map);
    final transcriptions =
        Map<String, dynamic>.from((form['transcriptions'] ?? {}) as Map);
    final pinyin = (transcriptions['pinyin'] ?? '').toString();
    final meanings = <String>[
      for (final m in (form['meanings'] as List? ?? const []))
        m.toString().trim(),
    ].where((m) => m.isNotEmpty).toList();
    if (meanings.isEmpty) continue;
    final pos = <String>[
      for (final p in (entry['pos'] as List? ?? const [])) p.toString().trim(),
    ].where((p) => p.isNotEmpty).toList();

    final id = _uniqueId(usedIds, pinyin, i);
    final encoded = Uri.encodeComponent(simplified);
    words.add({
      'id': id,
      'word': simplified,
      'phoneticUK': pinyin,
      'phoneticUS': pinyin,
      'audioUKUrl': 'https://dict.youdao.com/dictvoice?audio=$encoded&type=1',
      'audioUSUrl': 'https://dict.youdao.com/dictvoice?audio=$encoded&type=2',
      'partOfSpeech': pos.join('/'),
      'definitions': meanings,
      'englishDefinitions': meanings,
      'malayDefinitions': const <String>[],
      'exampleSentences': const <Map<String, String>>[],
      'collocations': const <Map<String, String>>[],
      'synonyms': const <String>[],
      'antonyms': const <String>[],
      'derivatives': const <String>[],
      'rootAffix': '',
      'examTags': const <String>[],
      'difficulty': 'Easy',
      'imageUrl': '',
      'hskLevel': _hskLevel,
    });
  }

  final book = {
    'id': 'hsk1',
    'title': 'HSK 1',
    'titleEn': 'HSK 1',
    'titleMs': 'HSK 1',
    'category': 'hsk',
    'examType': 'HSK 1',
    'coverUrl': '',
    'totalWords': words.length,
    'includedWords': words.length,
    'description': 'Core vocabulary for HSK Level 1.',
    'wordIds': [for (final w in words) w['id']],
    'createdAt': _createdAt,
    'words': words,
  };

  final encoder = const JsonEncoder.withIndent('  ');
  File(args[1]).writeAsStringSync('${encoder.convert(book)}\n');
  stdout.writeln('Wrote ${words.length} words to ${args[1]}');
}

String _uniqueId(Set<String> used, String pinyin, int index) {
  final ascii = pinyin
      .toLowerCase()
      .replaceAll(RegExp(r'[^a-z]'), '');
  final base = 'hsk1_${index.toString().padLeft(3, '0')}_${ascii.isEmpty ? 'x' : ascii}';
  var id = base;
  var suffix = 1;
  while (!used.add(id)) {
    id = '${base}_$suffix';
    suffix++;
  }
  return id;
}
```

```dart
class _TargetLanguageSetting extends StatelessWidget {
  const _TargetLanguageSetting({
    required this.targetLanguage,
    required this.onChanged,
  });

  final String targetLanguage;
  final ValueChanged<String> onChanged;

  // Active languages plus coming-soon placeholders.
  static const _active = {'zh': 'Chinese (中文)'};
  static const _comingSoon = {
    'es': 'Spanish (Español)',
    'fr': 'French (Français)',
    'ja': 'Japanese (日本語)',
  };

  @override
  Widget build(BuildContext context) {
    final theme = Theme.of(context);
    final label = _active[targetLanguage] ?? _active['zh']!;
    return AppCard(
      onTap: () => _showSheet(context),
      child: Row(
        children: [
          Icon(AppIcons.language, color: theme.colorScheme.primary),
          const SizedBox(width: AppTokens.s12),
          Expanded(
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Text('Target language', style: theme.textTheme.titleMedium),
                Text(
                  label,
                  style: TextStyle(color: theme.colorScheme.onSurfaceVariant),
                ),
              ],
            ),
          ),
          const Icon(AppIcons.chevronRight),
        ],
      ),
    );
  }

  void _showSheet(BuildContext context) {
    showModalBottomSheet(
      context: context,
      builder: (ctx) => SafeArea(
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            for (final entry in _active.entries)
              ListTile(
                leading: Icon(
                  entry.key == targetLanguage
                      ? AppIcons.radioButtonChecked
                      : AppIcons.radioButtonUnchecked,
                  color: Theme.of(ctx).colorScheme.primary,
                ),
                title: Text(entry.value),
                selected: entry.key == targetLanguage,
                onTap: () {
                  onChanged(entry.key);
                  Navigator.pop(ctx);
                },
              ),
            for (final entry in _comingSoon.entries)
              Opacity(
                opacity: 0.6,
                child: ListTile(
                  leading: const Icon(AppIcons.radioButtonUnchecked),
                  title: Text(entry.value),
                  trailing: const AppPill(
                    label: 'Coming Soon',
                    variant: PillVariant.neutral,
                  ),
                  enabled: false,
                ),
              ),
          ],
        ),
      ),
    );
  }
}
```
